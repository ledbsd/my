вручную

pyproject.toml

	[project]
	name = "note-app"
	version = "0.0.1"
	requires-python = ">=3.10"
	dependencies = ["requests"]

$ pip install .
$ pip install -e .

-------------------------------------------------------------
poetry

# сам создает pyproject.toml
$ poetry init

# если только управляем зависимостями, нужно добавить 
[tool.poetry]
package-mode = false

# добавить пакет
$ poetry add requests

# если нужно добавить в dev
$ poetry add --group dev requests

# обновить
$ poetry upgrade

# запустить в отдельном окружении
$ poetry run python main.py
